# Разметка диска
cfdisk /dev/sda
Стандартно - 1Гб на EFI, остальное на root. Своп - файлом.

Форматирование и монтирование
mkfs.ext4 /dev/sda2
mount /dev/sda2 /mnt

mkfs.fat -F32 /dev/sda1
mkdir -p /mnt/boot/efi	# устарело, теперь монтируется в /boot - mkdir /mnt/boot
mount /dev/sda1 /mnt/boot/efi	# соответственно, mount /dev/sda1 /mnt/boot

Swap потом в установленной системе делаем файлом.

# Загрузчик
Ставим grub2
pacman -S grub2 efibootmgr
grub-install /dev/sda
grub-mkconfig -o /boot/grub/grub.cfg

Или systemd-boot
bootctl install
Файл /boot/loader/loader.conf
default  arch.conf
timeout  4
console-mode max - максимальное разрешение экрана. Также есть варианты 0, 1, auto, keep.
Файл /boot/loader/entries/arch.conf
title   Arch Linux
linux   /vmlinuz-linux-zen
initrd  /intel-ucode.img
initrd  /initramfs-linux-zen.img
options root=UUID=12dad10e-2218-4796-ad89-2337030b4379 rw

Делаем пароль root
passwd

На этом этапе можно перезагрузить.

# Монтирование NTFS драйвером ядра (возможно какая-то хрень)
yay -S udisks2
В качестве обходного пути добавьте следующую опцию в секции [defaults] в файле /etc/udisks2/mount_options.conf:
ntfs_defaults=uid=$UID,gid=$GID,noatime,prealloc
